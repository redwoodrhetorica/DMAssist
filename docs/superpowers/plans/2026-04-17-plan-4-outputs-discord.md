# DM Assistant — Plan 4: Outputs & Discord

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the summary generator (Claude API → narrative prose), all output plugins (markdown file, PDF, clipboard), the "Generate Summary" button flow in the UI, and Discord integration as an advanced opt-in feature.

**Architecture:** Output plugins share an `OutputPlugin` interface and are invoked by the SummaryService after generation. The SummaryService calls the Claude API with the full session event log and character registry. PDF is generated by rendering an HTML template via puppeteer. Discord output uses discord.js and requires a bot token stored in the OS keychain.

**Tech Stack:** @anthropic-ai/sdk, puppeteer, discord.js, Electron clipboard API

**Prerequisite:** Plans 1, 2, and 3 complete. SessionStore, CampaignStore, SettingsStore, all IPC channels, and the full UI in place.

```ts
import { SessionEvent } from '../models/session'
import { Character } from '../models/campaign'
import { CampaignTone, SummaryLength } from '../models/settings'

export function buildStructuredLog(events: SessionEvent[], characters: Character[]): string {
  const charMap = Object.fromEntries(characters.map(c => [c.id, c.name]))

  return events
    .filter(e => e.inGame)
    .map(e => {
      const time = new Date(e.timestamp).toLocaleTimeString()
      const speaker = e.characterId ? (charMap[e.characterId] ?? e.speaker) : e.speaker
      return `[${time}] [${e.eventType.toUpperCase()}] ${speaker}: ${e.rawText}`
    })
    .join('\n')
}

export function buildSummaryPrompt(
  events: SessionEvent[],
  characters: Character[],
  tone: CampaignTone,
  length: SummaryLength
): string {
  const characterList = characters
    .filter(c => !c.archived)
    .map(c => `- ${c.name} (${c.characterClass} ${c.race}, played by ${c.playerName}): ${c.description}`)
    .join('\n')

  const structuredLog = buildStructuredLog(events, characters)

  const lengthInstructions: Record<SummaryLength, string> = {
    short: 'Write a short summary of 2-3 paragraphs covering only the most significant events.',
    medium: 'Write a medium-length summary of 4-6 paragraphs covering the main story beats and notable moments.',
    detailed: 'Write a detailed summary covering all significant events, character moments, and combat encounters.'
  }

  const toneInstructions: Record<CampaignTone, string> = {
    serious: 'Write in a serious, dramatic tone appropriate for an epic fantasy campaign journal.',
    heroic: 'Write in a heroic, adventurous tone that emphasizes the bravery and deeds of the characters.',
    humorous: 'Write in a light, humorous tone that captures the fun and comedic moments alongside the adventure.'
  }

  return `You are writing a campaign journal entry for a D&D session. Write in third person, past tense.

TONE: ${toneInstructions[tone]}

LENGTH: ${lengthInstructions[length]}

CHARACTERS IN THIS CAMPAIGN:
${characterList || 'No characters registered.'}

SESSION EVENT LOG (in-game events only, chronological):
${structuredLog || 'No events recorded.'}

Write the narrative summary now. Do not include any preamble or meta-commentary — just the journal entry itself.`
}
```

```ts
import Anthropic from '@anthropic-ai/sdk'
import { SessionEvent } from '../models/session'
import { Character } from '../models/campaign'
import { CampaignTone, SummaryLength } from '../models/settings'
import { SummaryPayload } from '../output/outputPlugin'
import { buildSummaryPrompt, buildStructuredLog } from './summaryPrompt'

const MODEL = 'claude-sonnet-4-6'

export interface GenerateOptions {
  sessionId: string
  campaignName: string
  events: SessionEvent[]
  characters: Character[]
  tone: CampaignTone
  length: SummaryLength
}

export class SummaryService {
  private client: Anthropic

  constructor(config: { apiKey: string }) {
    this.client = new Anthropic({ apiKey: config.apiKey })
  }

  async generate(options: GenerateOptions): Promise<SummaryPayload> {
    const { sessionId, campaignName, events, characters, tone, length } = options
    const prompt = buildSummaryPrompt(events, characters, tone, length)

    const response = await this.client.messages.create({
      model: MODEL,
      max_tokens: 2000,
      system: 'You are a campaign journal writer for tabletop RPG sessions. Write vivid, engaging prose.',
      messages: [{ role: 'user', content: prompt }]
    })

    const narrativeSummary = response.content[0].type === 'text' ? response.content[0].text : ''

    return {
      sessionId,
      campaignName,
      narrativeSummary,
      structuredLog: buildStructuredLog(events, characters),
      generatedAt: Date.now()
    }
  }
}
```

```ts
import { OutputPlugin, SummaryPayload } from './outputPlugin'

export class MarkdownOutput implements OutputPlugin {
  readonly name = 'markdown'

  constructor(private config: { folder: string; enabled: boolean }) {}

  isEnabled(): boolean {
    return this.config.enabled && !!this.config.folder
  }

  async deliver(payload: SummaryPayload): Promise<void> {
    const date = new Date(payload.generatedAt)
    const dateStr = date.toISOString().split('T')[0]
    const filename = `${dateStr}-${payload.sessionId.slice(0, 8)}.md`
    const filePath = path.join(this.config.folder, filename)

    fs.mkdirSync(this.config.folder, { recursive: true })

    const content = `# ${payload.campaignName} — Session Notes
*Generated: ${date.toLocaleString()}*

## Summary

${payload.narrativeSummary}

---

## Event Log

\`\`\`
${payload.structuredLog}
\`\`\`
`
    fs.writeFileSync(filePath, content, 'utf-8')
  }
}
```

```ts
import path from 'path'
import fs from 'fs'
import { OutputPlugin, SummaryPayload } from './outputPlugin'

export class PdfOutput implements OutputPlugin {
  readonly name = 'pdf'

  constructor(private config: { folder: string; enabled: boolean }) {}

  isEnabled(): boolean {
    return this.config.enabled && !!this.config.folder
  }

  async deliver(payload: SummaryPayload): Promise<void> {
    const puppeteer = await import('puppeteer')
    const browser = await puppeteer.default.launch({ headless: true })
    const page = await browser.newPage()

    const html = this.buildHtml(payload)
    await page.setContent(html, { waitUntil: 'networkidle0' })

    const date = new Date(payload.generatedAt).toISOString().split('T')[0]
    const filename = `${date}-${payload.sessionId.slice(0, 8)}.pdf`
    const filePath = path.join(this.config.folder, filename)

    fs.mkdirSync(this.config.folder, { recursive: true })
    await page.pdf({ path: filePath, format: 'A4', margin: { top: '40px', bottom: '40px', left: '50px', right: '50px' } })
    await browser.close()
  }

  private buildHtml(payload: SummaryPayload): string {
    const summary = payload.narrativeSummary.replace(/\n/g, '<br/>')
    const log = payload.structuredLog.replace(/\n/g, '<br/>')
    const date = new Date(payload.generatedAt).toLocaleString()

    return `<!DOCTYPE html>
<html>
<head>
<meta charset="utf-8">
<style>
  body { font-family: Georgia, serif; max-width: 700px; margin: 0 auto; color: #222; line-height: 1.7; }
  h1 { font-size: 24px; margin-bottom: 4px; }
  .meta { color: #888; font-size: 13px; margin-bottom: 32px; }
  h2 { font-size: 16px; text-transform: uppercase; letter-spacing: 1px; color: #555; margin-top: 32px; }
  .log { font-family: monospace; font-size: 12px; background: #f5f5f5; padding: 16px; border-radius: 4px; }
</style>
</head>
<body>
  <h1>${payload.campaignName} — Session Notes</h1>
  <div class="meta">Generated: ${date}</div>
  <h2>Summary</h2>
  <p>${summary}</p>
  <h2>Event Log</h2>
  <div class="log">${log}</div>
</body>
</html>`
  }
}
```

```ts
import { Client, GatewayIntentBits, TextChannel } from 'discord.js'
import { OutputPlugin, SummaryPayload } from './outputPlugin'

export class DiscordOutput implements OutputPlugin {
  readonly name = 'discord'

  constructor(private config: { botToken: string; channelId: string; enabled: boolean }) {}

  isEnabled(): boolean {
    return this.config.enabled && !!this.config.botToken && !!this.config.channelId
  }

  async deliver(payload: SummaryPayload): Promise<void> {
    const client = new Client({ intents: [GatewayIntentBits.Guilds] })

    await new Promise<void>((resolve, reject) => {
      client.once('ready', () => resolve())
      client.once('error', reject)
      client.login(this.config.botToken).catch(reject)
    })

    try {
      const channel = await client.channels.fetch(this.config.channelId)
      if (!channel || !(channel instanceof TextChannel)) {
        throw new Error(`Channel ${this.config.channelId} not found or not a text channel`)
      }

      const date = new Date(payload.generatedAt).toLocaleDateString()
      await channel.send({
        embeds: [{
          title: `${payload.campaignName} — Session Notes (${date})`,
          description: payload.narrativeSummary.slice(0, 4096),
          color: 0x5865F2,
          footer: { text: 'DM Assistant' }
        }]
      })
    } finally {
      client.destroy()
    }
  }
}
```

```ts
import { SummaryService } from '../summary/summaryService'
import { MarkdownOutput } from '../output/markdownOutput'
import { ClipboardOutput } from '../output/clipboardOutput'
import { PdfOutput } from '../output/pdfOutput'
import { DiscordOutput } from '../output/discordOutput'
import { SessionStore } from '../storage/sessionStore'
import { getSessionPath, getBasePath } from '../storage/paths'
import path from 'path'

ipcMain.handle('summary:generate', async (_e, campaignId: string, sessionId: string) => {
  const settings = settingsStore.load()
  const apiKey = await settingsStore.getSecret('claude-api-key')
  if (!apiKey) throw new Error('Claude API key not configured. Add it in Settings.')

  const campaign = campaignStore.loadCampaign(campaignId)
  if (!campaign) throw new Error(`Campaign ${campaignId} not found`)

  const dbPath = getSessionPath(campaignId, sessionId)
  const store = new SessionStore(dbPath)
  const events = store.getEvents(sessionId)
  store.close()

  const service = new SummaryService({ apiKey })
  const payload = await service.generate({
    sessionId,
    campaignName: campaign.name,
    events,
    characters: campaign.characters,
    tone: 'serious',
    length: 'medium'
  })

  const outputFolder = settings.output.markdownFolder || path.join(getBasePath(), 'summaries')

  const plugins = [
    new MarkdownOutput({ folder: outputFolder, enabled: settings.output.markdownEnabled }),
    new ClipboardOutput({ enabled: settings.output.clipboardEnabled }),
  ]

  if (settings.output.discordEnabled && settings.output.discordChannelId) {
    const botToken = await settingsStore.getSecret('discord-bot-token')
    if (botToken) {
      plugins.push(new DiscordOutput({
        botToken,
        channelId: settings.output.discordChannelId,
        enabled: true
      }))
    }
  }

  await Promise.all(
    plugins.filter(p => p.isEnabled()).map(p => p.deliver(payload))
  )

  return payload
})
```

```tsx
import { useState } from 'react'
import { Button } from '@/components/ui/button'
import { Sparkles, Loader2 } from 'lucide-react'
import SummaryModal from './SummaryModal'
import { ipc } from '../lib/ipc'
import { SummaryPayload } from '../../main/output/outputPlugin'

interface Props {
  campaignId: string
  sessionId: string | null
}

export default function GenerateSummaryButton({ campaignId, sessionId }: Props) {
  const [loading, setLoading] = useState(false)
  const [payload, setPayload] = useState<SummaryPayload | null>(null)

  const handleGenerate = async () => {
    if (!sessionId) return
    setLoading(true)
    try {
      const result = await ipc.summary.generate(campaignId, sessionId)
      setPayload(result)
    } catch (err) {
      alert(`Failed to generate summary: ${(err as Error).message}`)
    } finally {
      setLoading(false)
    }
  }

  return (
    <>
      <Button
        variant="outline"
        onClick={handleGenerate}
        disabled={!sessionId || loading}
        className="gap-2"
      >
        {loading ? <Loader2 className="h-4 w-4 animate-spin" /> : <Sparkles className="h-4 w-4" />}
        {loading ? 'Generating...' : 'Generate Summary'}
      </Button>
      <SummaryModal open={!!payload} payload={payload} onClose={() => setPayload(null)} />
    </>
  )
}
```

```tsx
import { useState } from 'react'
import { Button } from '@/components/ui/button'
import { Input } from '@/components/ui/input'
import { Label } from '@/components/ui/label'

interface Props {
  settings: AppSettings
  onUpdate: (settings: AppSettings) => void
}

export default function DiscordSetupWizard({ settings, onUpdate }: Props) {
  const [botToken, setBotToken] = useState('')
  const [channelId, setChannelId] = useState(settings.output.discordChannelId ?? '')
  const [tokenSaved, setTokenSaved] = useState(false)

    await ipc.settings.setSecret('discord-bot-token', botToken)
    setBotToken('')

    onUpdate({ ...settings, output: { ...settings.output, discordChannelId: channelId, discordEnabled: true } })
  }

  return (

          </div>
```
